rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Production rules: Each user can only access their own data
    match /users/{userId}/tasks/{taskId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // Allow users to read/write their user profile
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // Development fallback: shared user for localhost testing
    match /users/shared-user-upnext/tasks/{taskId} {
      allow read, write: if request.auth != null;
    }
    
    match /users/shared-user-upnext {
      allow read, write: if request.auth != null;
    }
    
    // Desktop collections: Allow authenticated users to access desktop-specific collections
    // This enables desktop app cross-authentication with anonymous Firebase users
    match /users/{userId}/tasks/{taskId} {
      allow read, write: if request.auth != null && userId.matches('^desktop-.*');
    }
    
    match /users/{userId} {
      allow read, write: if request.auth != null && userId.matches('^desktop-.*');
    }
  }
}
